from tkinter import *
from tkinter import ttk, messagebox
import mysql.connector


class ResultClass:
    def __init__(self, root):
        self.root = root
        self.root.title("Student Result Management System")
        self.root.geometry("1100x600+150+80")
        self.root.config(bg="white")
        self.root.focus_force()

        # ================= VARIABLES =================
        self.var_roll = StringVar()
        self.var_name = StringVar()
        self.var_course = StringVar()
        self.var_subject = StringVar()
        self.var_marks = StringVar()
        self.var_full_marks = StringVar()
        self.var_result = StringVar()
        self.var_search = StringVar()

        # ================= TITLE =================
        title = Label(
            self.root,
            text="Manage Student Result",
            font=("times new roman", 25, "bold"),
            bg="#0B4F9C",
            fg="white"
        )
        title.pack(side=TOP, fill=X)

        # ================= SEARCH FRAME =================
        search_frame = Frame(self.root, bg="white")
        search_frame.place(x=10, y=70, width=1080, height=60)

        Label(
            search_frame,
            text="Search By Roll No.",
            font=("times new roman", 15, "bold"),
            bg="white"
        ).place(x=10, y=15)

        Entry(
            search_frame,
            textvariable=self.var_search,
            font=("times new roman", 15),
            bg="lightyellow"
        ).place(x=180, y=12, width=250)

        Button(
            search_frame,
            text="Search",
            command=self.search,
            font=("times new roman", 13, "bold"),
            bg="#0B4F9C",
            fg="white",
            cursor="hand2"
        ).place(x=450, y=10, width=100, height=35)

        Button(
            search_frame,
            text="Show All",
            command=self.show,
            font=("times new roman", 13, "bold"),
            bg="#0B4F9C",
            fg="white",
            cursor="hand2"
        ).place(x=560, y=10, width=100, height=35)

        # ================= FORM =================
        form_frame = Frame(self.root, bd=2, relief=RIDGE, bg="white")
        form_frame.place(x=10, y=140, width=420, height=400)

        Label(
            form_frame,
            text="Student Result Details",
            font=("times new roman", 18, "bold"),
            bg="#0B4F9C",
            fg="white"
        ).pack(side=TOP, fill=X)

        # Roll No
        Label(
            form_frame,
            text="Roll No.",
            font=("times new roman", 14),
            bg="white"
        ).place(x=20, y=60)

        Entry(
            form_frame,
            textvariable=self.var_roll,
            font=("times new roman", 14),
            bg="lightyellow"
        ).place(x=150, y=58, width=230)

        # Name
        Label(
            form_frame,
            text="Student Name",
            font=("times new roman", 14),
            bg="white"
        ).place(x=20, y=105)

        Entry(
            form_frame,
            textvariable=self.var_name,
            font=("times new roman", 14),
            bg="lightyellow"
        ).place(x=150, y=103, width=230)

        # Course
        Label(
            form_frame,
            text="Course",
            font=("times new roman", 14),
            bg="white"
        ).place(x=20, y=150)

        Entry(
            form_frame,
            textvariable=self.var_course,
            font=("times new roman", 14),
            bg="lightyellow"
        ).place(x=150, y=148, width=230)

        # Subject
        Label(
            form_frame,
            text="Subject",
            font=("times new roman", 14),
            bg="white"
        ).place(x=20, y=195)

        Entry(
            form_frame,
            textvariable=self.var_subject,
            font=("times new roman", 14),
            bg="lightyellow"
        ).place(x=150, y=193, width=230)

        # Marks
        Label(
            form_frame,
            text="Marks Obtained",
            font=("times new roman", 14),
            bg="white"
        ).place(x=20, y=240)

        Entry(
            form_frame,
            textvariable=self.var_marks,
            font=("times new roman", 14),
            bg="lightyellow"
        ).place(x=150, y=238, width=230)

        # Full Marks
        Label(
            form_frame,
            text="Full Marks",
            font=("times new roman", 14),
            bg="white"
        ).place(x=20, y=285)

        Entry(
            form_frame,
            textvariable=self.var_full_marks,
            font=("times new roman", 14),
            bg="lightyellow"
        ).place(x=150, y=283, width=230)

        # Buttons
        Button(
            form_frame,
            text="Save",
            command=self.save,
            font=("times new roman", 13, "bold"),
            bg="#0B4F9C",
            fg="white",
            cursor="hand2"
        ).place(x=20, y=335, width=100, height=35)

        Button(
            form_frame,
            text="Update",
            command=self.update,
            font=("times new roman", 13, "bold"),
            bg="green",
            fg="white",
            cursor="hand2"
        ).place(x=140, y=335, width=100, height=35)

        Button(
            form_frame,
            text="Delete",
            command=self.delete,
            font=("times new roman", 13, "bold"),
            bg="red",
            fg="white",
            cursor="hand2"
        ).place(x=260, y=335, width=100, height=35)

        # ================= TABLE =================
        table_frame = Frame(self.root, bd=2, relief=RIDGE)
        table_frame.place(x=440, y=140, width=650, height=400)

        scroll_y = Scrollbar(table_frame, orient=VERTICAL)
        scroll_x = Scrollbar(table_frame, orient=HORIZONTAL)

        self.result_table = ttk.Treeview(
            table_frame,
            columns=(
                "roll",
                "name",
                "course",
                "subject",
                "marks",
                "full_marks"
            ),
            yscrollcommand=scroll_y.set,
            xscrollcommand=scroll_x.set
        )

        scroll_y.pack(side=RIGHT, fill=Y)
        scroll_x.pack(side=BOTTOM, fill=X)

        scroll_y.config(command=self.result_table.yview)
        scroll_x.config(command=self.result_table.xview)

        self.result_table.heading("roll", text="Roll No.")
        self.result_table.heading("name", text="Student Name")
        self.result_table.heading("course", text="Course")
        self.result_table.heading("subject", text="Subject")
        self.result_table.heading("marks", text="Marks")
        self.result_table.heading("full_marks", text="Full Marks")

        self.result_table["show"] = "headings"

        self.result_table.column("roll", width=80)
        self.result_table.column("name", width=130)
        self.result_table.column("course", width=100)
        self.result_table.column("subject", width=120)
        self.result_table.column("marks", width=80)
        self.result_table.column("full_marks", width=100)

        self.result_table.pack(fill=BOTH, expand=1)

        self.result_table.bind("<ButtonRelease-1>", self.get_data)

        self.show()

    # ================= DATABASE CONNECTION =================
    def connect_db(self):
        return mysql.connector.connect(
            host="localhost",
            user="root",
            password="user",
            database="srms"
        )

    # ================= SAVE =================
    def save(self):
        if self.var_roll.get() == "":
            messagebox.showerror("Error", "Roll No. is required")
            return

        try:
            con = self.connect_db()
            cur = con.cursor()

            cur.execute(
                """
                INSERT INTO result
                (roll, name, course, subject, marks, full_marks)
                VALUES (%s,%s,%s,%s,%s,%s)
                """,
                (
                    self.var_roll.get(),
                    self.var_name.get(),
                    self.var_course.get(),
                    self.var_subject.get(),
                    self.var_marks.get(),
                    self.var_full_marks.get()
                )
            )

            con.commit()
            con.close()

            messagebox.showinfo("Success", "Result saved successfully")
            self.clear()
            self.show()

        except Exception as ex:
            messagebox.showerror("Error", f"Error due to: {str(ex)}")

    # ================= SHOW =================
    def show(self):
        try:
            con = self.connect_db()
            cur = con.cursor()

            cur.execute("SELECT * FROM result")
            rows = cur.fetchall()

            self.result_table.delete(*self.result_table.get_children())

            for row in rows:
                self.result_table.insert("", END, values=row)

            con.close()

        except Exception as ex:
            messagebox.showerror("Error", f"Error due to: {str(ex)}")

    # ================= GET DATA =================
    def get_data(self, event):
        selected = self.result_table.focus()

        if selected == "":
            return

        data = self.result_table.item(selected)
        row = data["values"]

        if row:
            self.var_roll.set(row[0])
            self.var_name.set(row[1])
            self.var_course.set(row[2])
            self.var_subject.set(row[3])
            self.var_marks.set(row[4])
            self.var_full_marks.set(row[5])

    # ================= UPDATE =================
    def update(self):
        if self.var_roll.get() == "":
            messagebox.showerror("Error", "Select a result first")
            return

        try:
            con = self.connect_db()
            cur = con.cursor()

            cur.execute(
                """
                UPDATE result
                SET name=%s,
                    course=%s,
                    subject=%s,
                    marks=%s,
                    full_marks=%s
                WHERE roll=%s
                """,
                (
                    self.var_name.get(),
                    self.var_course.get(),
                    self.var_subject.get(),
                    self.var_marks.get(),
                    self.var_full_marks.get(),
                    self.var_roll.get()
                )
            )

            con.commit()
            con.close()

            messagebox.showinfo("Success", "Result updated successfully")
            self.clear()
            self.show()

        except Exception as ex:
            messagebox.showerror("Error", f"Error due to: {str(ex)}")

    # ================= DELETE =================
    def delete(self):
        if self.var_roll.get() == "":
            messagebox.showerror("Error", "Select a result first")
            return

        confirm = messagebox.askyesno(
            "Confirm",
            "Are you sure you want to delete this result?"
        )

        if confirm:
            try:
                con = self.connect_db()
                cur = con.cursor()

                cur.execute(
                    "DELETE FROM result WHERE roll=%s",
                    (self.var_roll.get(),)
                )

                con.commit()
                con.close()

                messagebox.showinfo("Success", "Result deleted successfully")
                self.clear()
                self.show()

            except Exception as ex:
                messagebox.showerror("Error", f"Error due to: {str(ex)}")

    # ================= SEARCH =================
    def search(self):
        if self.var_search.get() == "":
            self.show()
            return

        try:
            con = self.connect_db()
            cur = con.cursor()

            cur.execute(
                "SELECT * FROM result WHERE roll=%s",
                (self.var_search.get(),)
            )

            rows = cur.fetchall()

            self.result_table.delete(*self.result_table.get_children())

            for row in rows:
                self.result_table.insert("", END, values=row)

            con.close()

        except Exception as ex:
            messagebox.showerror("Error", f"Error due to: {str(ex)}")

    # ================= CLEAR =================
    def clear(self):
        self.var_roll.set("")
        self.var_name.set("")
        self.var_course.set("")
        self.var_subject.set("")
        self.var_marks.set("")
        self.var_full_marks.set("")


if __name__ == "__main__":
    root = Tk()
    obj = ResultClass(root)
    root.mainloop()
