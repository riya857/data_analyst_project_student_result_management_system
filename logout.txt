import mysql.connector
import os
import time


class RMS:

    def __init__(self, root):

        self.root = root

        self.root.title("Student Result Management System")
        self.root.geometry("1200x700+50+30")
        self.root.config(bg="white")

        # =========================================================
        # TITLE
        # =========================================================

        title = Label(
            self.root,
            text="Student Result Management System",
            font=("times new roman", 32, "bold"),
            bg="#263238",
            fg="white"
        )

        title.place(
            x=0,
            y=0,
            relwidth=1,
            height=80
        )

        # MENU FRAME
        # =========================================================

            x=0,
            y=90,
            relwidth=1,
            height=70
        )

        # =========================================================
        # BUTTON FUNCTION
        # =========================================================

        def open_file(filename):

            current_folder = os.path.dirname(
                os.path.abspath(__file__)
            )

            possible_files = [
                filename,
                filename + ".py"
            ]

            found_file = None

            for file in possible_files:

                path = os.path.join(
                    current_folder,
                    file
                )

                if os.path.exists(path):
                    found_file = path
                    break

            if found_file:

                try:

                    os.system(
                        f'python "{found_file}"'
                    )

                except Exception as e:

                    messagebox.showerror(
                        "Error",
                        str(e)
                    )

            else:

                messagebox.showerror(
                    "File Not Found",
                    f"{filename} file nahi mili."
                )

        # =========================================================
        # COURSE BUTTON
        # =========================================================

        course_btn = Button(
            menu_frame,
            text="Course",
            font=("times new roman", 16, "bold"),
            bg="#263B73",
            fg="white",
            cursor="hand2",
            command=lambda: open_file("course")
        )

        course_btn.place(
            x=20,
            y=5,
            width=180,

        # VIEW STUDENT RESULT
        # =========================================================

        view_result_btn = Button(

            text="View Student Result",
            font=("times new roman", 16, "bold"),
            bg="#263B73",
            fg="white",
            cursor="hand2",
            command=lambda: open_file("view student result")
        )

        view_result_btn.place(
            x=620,
            y=5,
            width=220,
            height=55
        )

        # =========================================================
        # LOGOUT BUTTON
        # =========================================================

        logout_btn = Button(
            menu_frame,

            font=("times new roman", 16, "bold"),
            bg="#263B73",
            fg="white",
            cursor="hand2",
            command=self.logout
        )

        logout_btn.place(
            x=860,
            y=5,
            width=150,
            height=55
        )

        # =========================================================
        # EXIT BUTTON
        # =========================================================

        exit_btn = Button(
            menu_frame,

            font=("times new roman", 16, "bold"),

            fg="white",
            cursor="hand2",
            command=self.exit_app

            y=5,

            height=55
        )

        # =========================================================
        # MAIN IMAGE FRAME
        # =========================================================

            self.root,
            bg="white",

            relief=RIDGE

            x=450,

            width=735,
            height=390
        )

        # =========================================================
        # DASHBOARD IMAGE
        # =========================================================

        try:

            current_folder = os.path.dirname(
                os.path.abspath(__file__)
            )

            image_path = None

            # images.jpg ko automatically search karega
            for root_dir, dirs, files in os.walk(current_folder):

                for file in files:

                    if file.lower() == "images.jpg":

                        image_path = os.path.join(
                            root_dir,
                            file
                        )

                        break

                if image_path:
                    break

            if image_path:

                print("Dashboard Image Found:")
                print(image_path)

                img = img.resize(
                    (730, 385),
                    Image.Resampling.LANCZOS
                )

                self.dashboard_image = ImageTk.PhotoImage(
                    img
                )

                    self.image_frame,
                    image=self.dashboard_image,
                    bg="white"
                )

                    fill=BOTH,
                    expand=True
                )

                    self.image_frame,

                        "bold"

                    bg="white"

                    fill=BOTH,
                    expand=True

                self.image_frame,
                text="Dashboard Image Not Found",

                    "times new roman",

                    "bold"

            text="[ 0 ]",
            font=("times new roman", 18, "bold"),

            fg="white"
        )

    # =============================================================
    # CLOCK
    # =============================================================

    def update_clock(self):

        current_time = time.strftime(
            "%I:%M:%S %p"
        )

        self.clock_label.config(
            text=current_time
        )

        self.root.after(
            1000,
            self.update_clock
        )

    # =============================================================
    # DATABASE COUNTS
    # =============================================================

    def update_counts(self):

        try:

            con = mysql.connector.connect(
                host="localhost",
                user="root",
                password="user",
                database="rms"
            )

            cursor = con.cursor()

            # Student count
            try:

                cursor.execute(
                    "SELECT COUNT(*) FROM student"
                )

                student_count = cursor.fetchone()[0]

                self.student_count_label.config(
                    text=f"[ {student_count} ]"
                )

            except:

                self.student_count_label.config(
                    text="[ 0 ]"
                )

            # Course count
            try:

                cursor.execute(
                    "SELECT COUNT(*) FROM course"
                )

                course_count = cursor.fetchone()[0]

                self.course_count_label.config(
                    text=f"[ {course_count} ]"
                )

            except:

                self.course_count_label.config(
                    text="[ 0 ]"
                )

            # Result count
            try:

                cursor.execute(
                    "SELECT COUNT(*) FROM result"
                )

                result_count = cursor.fetchone()[0]

                self.result_count_label.config(
                    text=f"[ {result_count} ]"
                )

            except:

                self.result_count_label.config(
                    text="[ 0 ]"
                )

            cursor.close()
            con.close()

        except Exception as e:

            print("Database Error:", e)

    # =============================================================
    # LOGOUT
    # =============================================================

    def logout(self):

        answer = messagebox.askyesno(
            "Logout",
            "Are you sure you want to logout?"
        )

        if answer:

            self.root.destroy()

            # Agar login.py hai to ye line uncomment kar dena:
            # os.system("python login.py")

    # =============================================================
    # EXIT
    # =============================================================

    def exit_app(self):

        answer = messagebox.askyesno(
            "Exit",
            "Are you sure you want to exit?"
        )

        if answer:

            self.root.destroy()
